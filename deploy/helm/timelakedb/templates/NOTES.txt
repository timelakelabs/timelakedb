  curl -fs http://<svc>:{{ .Values.service.httpPort }}/health

  # write a point, then read it back
  curl -fs -XPOST "http://<svc>:{{ .Values.service.httpPort }}/api/v3/write_lp?db=poc&precision=ns" \
    --data-binary 'demo,host=a value=1 1'
  curl -fs -XPOST "http://<svc>:{{ .Values.service.httpPort }}/api/sql" \
    -H 'content-type: application/json' \
    -d '{"db":"poc","sql":"SELECT COUNT(*) AS n FROM demo"}'
